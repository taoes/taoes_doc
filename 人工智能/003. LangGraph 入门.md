```python
import json
import operator
import os
from typing import List, TypedDict, Annotated, Sequence

from dotenv import load_dotenv
from langchain.chat_models import init_chat_model
from langchain_core.messages import SystemMessage, HumanMessage, BaseMessage
from langgraph.graph import StateGraph, START, END
from pydantic import BaseModel

load_dotenv()

chat_model = init_chat_model(
    model='deepseek-v4-pro',
    api_key=os.getenv('DEEPSEEK_API_KEY'),
    temperature=0.5
)


class State(TypedDict):
    raw_input: str  # 用户原始输入
    messages: Annotated[Sequence[BaseMessage], operator.add]  # 消息累加
    response: str  # 最终输出结果


def supervisor_choose(s: State):
    system = '''
         你是一个专业的用户意图分析师，可以根据用户的输入返回具体执行的意图，目前支持已下意图
         - QUERY: 咨询查询类
         - WRITE: 写作类
         - CHAT: 闲聊类
         - NONE: 未识别到意图
         如果没有识别到用户的意图或者和已知的意图不匹配，则返回 NONE，不要随意没有凭证的返回用户的信息
         仅仅返回上面的枚举即可，不需要额外的解释或者思考内容
         '''
    messages = [
        SystemMessage(system),
        HumanMessage(s['raw_input']),
    ]
    resp = chat_model.invoke(messages)
    print(resp.content)
    return resp.content


def supervisor(s: State):
    return s


def query(s: State):
    system = '''
        你是一个专业的信息检索师，可以根据用户的问题去查询资料
    '''
    messages = [
        SystemMessage(system),
        HumanMessage(s['raw_input']),
    ]
    resp = chat_model.invoke(messages)
    print(resp.content)
    return {
        "messages": s['messages'] + messages,
        "response": resp.content,
    }


def chat(s: State):
    system = '''
        你是一个专业的情感陪伴式，可以根据用户的问题和要求聊天，要求提供最大的情绪价值
    '''
    messages = [
        SystemMessage(system),
        HumanMessage(s['raw_input']),
    ]
    resp = chat_model.invoke(messages)
    print(resp.content)
    return {
        "messages": s['messages'] + messages,
        "response": resp.content,
    }


def write(s: State):
    system = '''
        你是一个专业的作家，可以根据用户的要求实现写诗或者写一篇短文
    '''
    messages = [
        SystemMessage(system),
        HumanMessage(s['raw_input']),
    ]
    resp = chat_model.invoke(messages)
    print(resp.content)
    return {
        "messages": s['messages'] + messages,
        "response": resp.content,
    }


if __name__ == '__main__':
    build = StateGraph(State)
    build.add_node("supervisor", supervisor)
    build.add_node("query", query)
    build.add_node("write", write)
    build.add_node("chat", chat)

    build.add_edge(START, 'supervisor')
    build.add_conditional_edges('supervisor', supervisor_choose, {
        'QUERY': 'query',
        'WRITE': 'write',
        'CHAT': 'chat',
        'NONE': END,
        END: END
    })
    # 查询/写作/闲聊执行完成后统一走向结束
    build.add_edge("query", END)
    build.add_edge("write", END)
    build.add_edge("chat", END)

    graph = build.compile()
    graph.get_graph().draw_mermaid_png(output_file_path='./draw.png')
    input_payload: State = {
        "raw_input": "写一段 50 字的英文短文，描述中学生李华的幸福生活",
        "messages": [],
        "response": ""
    }
    resp = graph.invoke(input_payload)
    print("\n====最终回答====")
    print(resp['response'])
```
