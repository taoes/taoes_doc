#include <stdlib.h>
#include "stdio.h"



int main(int argc,char* args[]){
    char* path = args[1];
    printf("class file path = %s\r\n",path);

    FILE* file;
    file = fopen(path,"r");
    if(file == NULL){
        perror(path);
        exit(-1);
    }
    unsigned  char buf[10];
    int rc;
    while (1){
        rc = fread(buf,1,1,file);
        if(rc <= 0){
            break;
        }
        printf("%2X ",buf[0]);
    }


    return 0;

}
