```c
#include <stdlib.h>
#include "stdio.h"


// 定义结构体
struct Student {
    char *name;
    int age;
    char *address;
};

void print(struct Student *stu) {
    printf("name = %s,age = %d,address = %s\n", stu->name, stu->age, stu->address);
    stu->name = "李四";
}

int main(int argc, char *args[]) {
    struct Student stu;
    stu.name = "张三";
    stu.age = 18;
    stu.address = "上海市杨浦区";
    print(&stu);
    print(&stu);

}
```
